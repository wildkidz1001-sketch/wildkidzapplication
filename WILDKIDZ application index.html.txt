    function validateStage(stageIndex) {

        if (selectedRoles.size === 0) {

          return false;

        return true;

      const section = document.querySelector(`.stage-view[data-stage="${stageIndex}"]`);
      if (!section) return true;


      const requiredFields = section.querySelectorAll('[required]');
      for (const field of requiredFields) {
        if (!field.value.trim()) {
          field.focus();
          showToast('Please complete all required fields in this stage before continuing.');
          return false;
        }
        if (field.type === 'email' && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(field.value.trim())) {
          field.focus();
          showToast('Please enter a valid email address so WOWLD Entertainment can contact you.');
          return false;
        }
      }


      if (stageIndex === 1) {
        const ageVal = parseInt(document.getElementById('age').value, 10);
        if (isNaN(ageVal) || ageVal < 10 || ageVal > 45) {
          document.getElementById('age').focus();
          showToast('Please enter a valid age.');
          return false;
        }
      }


      if (stageIndex === 12) {
        const ageVal = parseInt(document.getElementById('age').value, 10);

          return false;

      return true;

    function toggleRole(tile) {
      const role = tile.getAttribute('data-role');
      if (selectedRoles.has(role)) {
        selectedRoles.delete(role);
        tile.classList.remove('selected');

    function renderMediaPreview(type, objectUrl) {
      const box = document.getElementById(`preview-${type}`);
      const removeBtn = document.getElementById(`remove-${type}`);
      const state = mediaState[type];


      if (!state.link && !state.fileName) {
        box.style.display = 'none';
        removeBtn.style.display = 'none';
        box.innerHTML = '';
        return;
      }


      removeBtn.style.display = 'inline-block';
      box.style.display = 'block';
      let html = '';


      if (state.fileName && objectUrl) {
        const isVideo = state.fileObj && state.fileObj.type.startsWith('video/');
        html += `<div class="mono" style="font-size:10px; color:#FFFFFF;">ATTACHED FILE: ${escapeHtml(state.fileName)}</div>`;
        html += isVideo
          ? `<video controls src="${objectUrl}"></video>`
          : `<audio controls src="${objectUrl}"></audio>`;
      }


      if (state.link) {
        html += `<div class="mono" style="font-size:10px; color:var(--silver-300); margin-top:6px;">EXTERNAL LINK: <a href="${escapeHtml(state.link)}" target="_blank" rel="noopener" style="color:#FFFFFF;">${escapeHtml(state.link)}</a></div>`;
      }


      box.innerHTML = html;
    }

    const canvas = document.getElementById('starfieldCanvas');
    const ctx = canvas.getContext('2d');
    let stars = [];


    function resizeCanvas() {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    }


    function createStars() {
      const count = Math.min(48, Math.floor((window.innerWidth * window.innerHeight) / 24000));
      stars = Array.from({ length: count }, () => {
        const isCrimson = Math.random() < 0.22;
        return {
          x: Math.random() * canvas.width,
          y: Math.random() * canvas.height,
          size: 4 + Math.random() * 14,
          vx: (Math.random() - 0.5) * 0.28,
          vy: -0.15 - Math.random() * 0.35,
          rot: Math.random() * Math.PI,
          vRot: (Math.random() - 0.5) * 0.012,
          alpha: 0.32 + Math.random() * 0.55,
          crimson: isCrimson
        };
      });
    }


    function drawFourPointStar(cx, cy, r, rot, color, glow) {
      ctx.save();
      ctx.translate(cx, cy);
      ctx.rotate(rot);
      ctx.beginPath();
      const inner = r * 0.2;
      ctx.moveTo(0, -r);
      ctx.quadraticCurveTo(inner * 0.4, -inner * 0.4, r, 0);
      ctx.quadraticCurveTo(inner * 0.4, inner * 0.4, 0, r);
      ctx.quadraticCurveTo(-inner * 0.4, inner * 0.4, -r, 0);
      ctx.quadraticCurveTo(-inner * 0.4, -inner * 0.4, 0, -r);
      ctx.closePath();
      ctx.shadowBlur = 14;
      ctx.shadowColor = glow;
      ctx.fillStyle = color;
      ctx.fill();
      ctx.restore();
    }


    function animateStars() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      for (const s of stars) {
        s.x += s.vx;
        s.y += s.vy;
        s.rot += s.vRot;
        if (s.y < -30) {
          s.y = canvas.height + 30;
          s.x = Math.random() * canvas.width;
        }
        if (s.x < -30) s.x = canvas.width + 30;
        if (s.x > canvas.width + 30) s.x = -30;


        const fill = s.crimson
          ? `rgba(225, 29, 72, ${s.alpha})`
          : `rgba(226, 232, 240, ${s.alpha})`;
        const glow = s.crimson
          ? 'rgba(225, 29, 72, 0.55)'
          : 'rgba(255, 255, 255, 0.38)';
        drawFourPointStar(s.x, s.y, s.size, s.rot, fill, glow);
      }
      requestAnimationFrame(animateStars);
    }


    window.addEventListener('resize', () => {
      resizeCanvas();
      createStars();
    });
